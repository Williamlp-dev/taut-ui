## Usage

```tsx
import { Button } from "@/components/ui/button"
import { Field, FieldError, FieldLabel } from "@/components/ui/field"
import { Form } from "@/components/ui/form"
import { Input } from "@/components/ui/input"
```

```tsx
<Form onSubmit={(e) => e.preventDefault()}>
  <Field name="email">
    <FieldLabel>Email</FieldLabel>
    <Input required type="email" />
    <FieldError />
  </Field>
  <Button type="submit">Submit</Button>
</Form>
```

## API Reference

### Form

Wraps Base UI `Form` (`<form>`).

| Prop             | Type                                                         | Default      | Description                                        |
| ---------------- | ------------------------------------------------------------ | ------------ | -------------------------------------------------- |
| `errors`         | `Record<string, string \| string[]>`                         | -            | External/server validation errors keyed by `name`  |
| `validationMode` | `"onSubmit" \| "onBlur" \| "onChange"`                       | `"onSubmit"` | Controls when field validation triggers            |
| `onFormSubmit`   | `(formValues: Record<string, unknown>, details) => void`     | -            | Called on valid submission (`preventDefault` built-in) |
| `actionsRef`     | `React.RefObject<Form.Actions \| null>`                      | -            | Imperative handle exposing `validate(fieldName?)`  |
| `className`      | `string`                                                     | -            | Additional CSS classes                             |
