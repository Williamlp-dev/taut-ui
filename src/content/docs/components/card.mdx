## Usage

```tsx
import {
  Card,
  CardHeader,
  CardTitle,
  CardDescription,
  CardAction,
  CardContent,
  CardFooter,
  CardMedia,
} from "@/components/ui/card"
```

```tsx
<Card variant="elevated">
  <CardMedia>
    <img src="/hero.jpg" alt="Hero" />
  </CardMedia>
  <CardHeader>
    <CardTitle>Card Title</CardTitle>
    <CardDescription>A short description of this card.</CardDescription>
    <CardAction>{/* optional action, e.g. a button or badge */}</CardAction>
  </CardHeader>
  <CardContent>
    <p>Main body content goes here.</p>
  </CardContent>
  <CardFooter>
    <button>Cancel</button>
    <button>Confirm</button>
  </CardFooter>
</Card>
```

### Horizontal card

```tsx
<Card orientation="horizontal" variant="outlined">
  <CardMedia>
    <img src="/thumbnail.jpg" alt="Thumbnail" />
  </CardMedia>
  <CardHeader>
    <CardTitle size="md">Horizontal Card</CardTitle>
    <CardDescription>Media aligns to the left at a fixed width.</CardDescription>
  </CardHeader>
</Card>
```

### Card Frame (grouped cards)

Use `CardFrame` to visually group multiple `Card` items into a single container.

```tsx
import {
  CardFrame,
  CardFrameHeader,
  CardFrameTitle,
  CardFrameDescription,
  CardFrameAction,
  CardFrameContent,
  CardFrameFooter,
  Card,
  CardHeader,
  CardTitle,
  CardContent,
} from "@/components/ui/card"

<CardFrame>
  <CardFrameHeader>
    <CardFrameTitle>Group Title</CardFrameTitle>
    <CardFrameDescription>A collection of related cards.</CardFrameDescription>
  </CardFrameHeader>
  <Card>
    <CardHeader>
      <CardTitle>Item One</CardTitle>
    </CardHeader>
    <CardContent>Content for item one.</CardContent>
  </Card>
  <Card>
    <CardHeader>
      <CardTitle>Item Two</CardTitle>
    </CardHeader>
    <CardContent>Content for item two.</CardContent>
  </Card>
  <CardFrameFooter>
    <span>Footer info</span>
    <button>See all</button>
  </CardFrameFooter>
</CardFrame>
```

## API

### `Card`

The root card surface. Renders a `<div>` by default; pass `render={<button />}` or `render={<a />}` to make it interactive.

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `variant` | `"elevated" \| "filled" \| "outlined"` | `"elevated"` | Visual style of the card. `elevated` adds a drop shadow; `filled` uses the muted background; `outlined` shows a border. |
| `orientation` | `"vertical" \| "horizontal"` | `"vertical"` | Layout direction. `horizontal` arranges children in a row and reduces internal padding. |
| `render` | `React.ReactElement \| ((props, ref) => React.ReactElement)` | `<div>` | Override the rendered element (e.g. `<button>` or `<a>`). Inherited from Base UI `useRender`. |
| `className` | `string` | — | Additional CSS classes merged with the variant classes. |
| `...props` | `React.HTMLAttributes<HTMLDivElement>` | — | All other props are forwarded to the underlying element. |

---

| Prop | Type | Default | Description |

---

### `CardHeader`

| `...props` | `React.HTMLAttributes<HTMLDivElement>` | — | Forwarded to the underlying element. |

### `CardTitle`

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `size` | `"md" \| "lg"` | `"md"` | `"md"` renders at `text-lg`; `"lg"` renders at `text-xl`. In horizontal cards the size is always reduced to `text-base`. |
| `render` | `React.ReactElement \| ((props, ref) => React.ReactElement)` | `<div>` | Override the rendered element. |
| `className` | `string` | — | Additional CSS classes. |
| `...props` | `React.HTMLAttributes<HTMLDivElement>` | — | Forwarded to the underlying element. |
