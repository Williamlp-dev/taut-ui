## Usage

```tsx
import { Button } from "@/components/ui/button"
import {
  Empty,
  EmptyContent,
  EmptyDescription,
  EmptyHeader,
  EmptyMedia,
  EmptyTitle,
} from "@/components/ui/empty"
```

```tsx
<Empty>
  <EmptyMedia>
    <InboxIcon />
  </EmptyMedia>
  <EmptyHeader>
    <EmptyTitle>No messages yet</EmptyTitle>
    <EmptyDescription>
      When you receive messages, they will appear here.
    </EmptyDescription>
  </EmptyHeader>
  <EmptyContent>
    <Button>Compose message</Button>
  </EmptyContent>
</Empty>
```

## API Reference

### Empty

Root centered flex-column container for the empty state.

| Prop        | Type                          | Description                               |
| ----------- | ----------------------------- | ----------------------------------------- |
| `className` | `string`                      | Additional CSS classes                    |
| `...props`  | `React.ComponentProps<"div">` | All standard div attributes are supported |

### EmptyMedia

Visual media container displayed above `EmptyHeader`, featuring a layered squircle backdrop for icon variants.

| Prop        | Type                              | Default  | Description                                                                   |
| ----------- | --------------------------------- | -------- | ----------------------------------------------------------------------------- |
| `variant`   | `"icon" \| "vibrant" \| "default"` | `"icon"` | `"icon"` uses neutral surfaces, `"vibrant"` uses primary, `"default"` is bare |
| `className` | `string`                          | -        | Additional CSS classes                                                        |
| `...props`  | `React.ComponentProps<"div">`     | -        | All standard div attributes are supported                                     |

### EmptyHeader

Groups `EmptyTitle` and `EmptyDescription` with centered text alignment.

### EmptyTitle

Heading (`<h3>`) for the empty-state title.

### EmptyDescription

Supporting paragraph (`<p>`) with balanced line width (`max-w-[36ch]`) and styled inline links.

### EmptyContent

Flexible row container for primary actions, secondary buttons, or inline inputs.
